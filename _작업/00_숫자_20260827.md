## 세는 방법

```bash
cd ~/workspace/움직이는그림사전

import csv,json
m={r['단어'].strip().lower() for r in csv.DictReader(open('data/1200_마스터리스트_v1.csv',encoding='utf-8-sig')) if r.get('단어')}

used=set(); n=0
for v in data.values():
    for w in v['works']:
        n+=1; used |= {p[0].strip().lower() for p in w.get('words',[])}
print(f"마스터 {len(m)} / 배정 {len(used)} / 미배정 {len(m-used)} / {n}편 / 편당 {len(used)/n:.1f}")
```
