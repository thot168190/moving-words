### 3. 기계 검증

```bash
cd ~/workspace/움직이는그림사전
python3 - <<'PY'
import json
s=open('public/learning/index.html',encoding='utf-8').read()
st=s.find('const chapterData = '); b=s.find('{',st); d=0
for k in range(b,len(s)):
    if s[k]=='{': d+=1
    elif s[k]=='}':
        d-=1
        if d==0: break
D=json.loads(s[b:k+1])
bad=[]
for c,v in D.items():
    for i,w in enumerate(v['works']):
        l1,l2 = v['levelOneWords'][i], v['levelTwoWords'][i]
        s1,s2 = v['levelOneSpots'][i], v['sceneSpots'][i]
        if len(l1)!=len(s1) or len(l2)!=len(s2):
            bad.append(f"{c}-{w['n']} 좌표 불일치")
        if len(w['words'])!=len(l1)+len(l2):
            bad.append(f"{c}-{w['n']} words 개수 불일치")
        if len(l1)+len(l2) > 8:
            bad.append(f"{c}-{w['n']} 8개 초과")
print(bad or "이상 없음")
PY
```
